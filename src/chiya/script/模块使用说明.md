# token解析/词法解析
```java
 ChiyaTokenParser chiyaTokenParser = new ChiyaTokenParser();
 // 单个字符
 chiyaTokenParser.addToken("token", ".", "(", ")", "[", "]", "{", "}", "@", "?", ":", ",");
 chiyaTokenParser.addToken("空白符", " ", "\t", "\n");
 // 成对字符
 chiyaTokenParser.addCombination("注释", "/*", "*/");
 chiyaTokenParser.addCombination("注释", "//", "\n");
 // 输出token
 // 第一个参数是原始字符串序列，第二个参数为未匹配到的类型，第三个起始行坐标，第四个忽略的类型，可以填多个参数
 List<ChiyaToken list = chiyaTokenParser.toToken(code, "any", 0, "空白符");
```


# 语法解析
```java
 ChiyaSyntaxParser chiyaSyntaxParser = new ChiyaSyntaxParser();
 // 标记关键字，会直接将data等于该值的标记为key:xxxx
 chiyaSyntaxParser.registerKeyword("if", "else", "public", "private");
 // 标记token序列
 chiyaSyntaxParser.markKeyword(list);

 // 解析器配置，详细见内部参数
 ParserConfig config;
 // 语法配置，详细见内部参数
 SyntaxFactor syntax;
 
 config = new ParserConfig();
 syntax = new SyntaxFactor("else_if")
 	.addLexical(new LexicalFactor().chainType("key:else"))
	.addLexical(new LexicalFactor().chainType("key:if"));
 chiyaSyntaxParser.addFlow(0, config, syntax);

 config = new ParserConfig().chainNeedRecursion(true);
 syntax = new SyntaxFactor("括号")
	.addLexical(new LexicalFactor().chainType("token").chainStart("("))
	.addLexical(new LexicalFactor().chainType("token").chainStart(")"))
	.chainNeedMatch(true)
	.chainNeedPaired(true);
 chiyaSyntaxParser.addFlow(1, config, syntax);

 config = new ParserConfig();
 syntax = new SyntaxFactor("修饰符")
	.chainNeedMerge(true)
	.chainFatherIndex(0)
	.addLexical(new LexicalFactor().chainType("key:public", "key:static", "key:final", "key:void", "修饰符"))
	.addLexical(new LexicalFactor().chainType("key:public", "key:static", "key:final", "key:void"));
 chiyaSyntaxParser.addFlow(50.001, config, syntax);


 System.out.println(ChiyaToken.debug(chiyaSyntaxParser.parser(list)));
```
